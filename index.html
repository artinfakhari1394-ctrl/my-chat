-- ========================================= -- CHAT APP DATABASE - COMPLETE SQL -- ========================================= -- 1. Create profiles table CREATE TABLE IF NOT EXISTS public.profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, username TEXT NOT NULL UNIQUE CHECK (username ~ '^[A-Za-z0-9_]{3,20}$'), display_name TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 2. Create messages table CREATE TABLE IF NOT EXISTS public.messages ( id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, sender_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE, recipient_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE, content TEXT NOT NULL CHECK (CHAR_LENGTH(content) BETWEEN 1 AND 4000), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), CONSTRAINT different_users CHECK (sender_id <> recipient_id) ); -- 3. Enable Row Level Security ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE public.messages ENABLE ROW LEVEL SECURITY; -- 4. Remove old policies if they already exist DROP POLICY IF EXISTS "Authenticated users can view profiles" ON public.profiles; DROP POLICY IF EXISTS "Users can read their own conversations" ON public.messages; DROP POLICY IF EXISTS "Users can send messages as themselves" ON public.messages; -- 5. Allow logged-in users to view profiles CREATE POLICY "Authenticated users can view profiles" ON public.profiles FOR SELECT TO authenticated USING (TRUE); -- 6. Allow users to read messages they sent or received CREATE POLICY "Users can read their own conversations" ON public.messages FOR SELECT TO authenticated USING ( auth.uid() = sender_id OR auth.uid() = recipient_id ); -- 7. Allow users to send messages as themselves CREATE POLICY "Users can send messages as themselves" ON public.messages FOR INSERT TO authenticated WITH CHECK ( auth.uid() = sender_id AND EXISTS ( SELECT 1 FROM public.profiles AS p WHERE p.id = recipient_id ) ); -- 8. Create automatic profile registration function CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ BEGIN INSERT INTO public.profiles ( id, username, display_name ) VALUES ( NEW.id, NEW.raw_user_meta_data ->> 'username', NEW.raw_user_meta_data ->> 'display_name' ); RETURN NEW; END; $$; -- 9. Recreate registration trigger DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users; CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.handle_new_user(); -- 10. Grant required permissions GRANT SELECT ON public.profiles TO authenticated; GRANT SELECT, INSERT ON public.messages TO authenticated; GRANT USAGE, SELECT ON SEQUENCE public.messages_id_seq TO authenticated; -- 11. Create indexes for faster message loading CREATE INDEX IF NOT EXISTS idx_messages_sender_recipient ON public.messages (sender_id, recipient_id, created_at); CREATE INDEX IF NOT EXISTS idx_messages_recipient_sender ON public.messages (recipient_id, sender_id, created_at); CREATE INDEX IF NOT EXISTS idx_profiles_username ON public.profiles (username); -- ========================================= -- DONE -- =========================================
